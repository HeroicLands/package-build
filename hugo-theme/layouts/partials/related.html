{{- /* Related partial. Renders a compact "Related" block at the bottom of a single page from the page's `related` front matter, if it carries any. Structure in front matter: related: seeAlso: [{ title, url, type, gloss? }, ...] # deliberate references backlinks: [{ title, url, type }, ...] # pages that link here mentions: [{ title, url, type }, ...] # pages this page links to Rendering strategy: - Render deliberate references in authored order, with their glosses. - Merge both inferred directions into a single deduped pool (direction doesn't matter to a reader — a connection is a connection). - Group by the referenced page's `type`. - Emit one labelled group per type, its entries laid out as a CSS grid (`repeat(auto-fill, minmax(14em, 1fr))`) so a wide screen shows two or three columns and a narrow one falls back to a single column with no media query. - The whole block is wrapped in a bordered, tinted card so it visually sets off the end of the article. Silent disappear: a page with no `related`, or whose lists are all empty, emits nothing at all rather than an empty heading. The rows are computed first and the
is rendered only when there is something to put in it. `return` cannot bail out early here. It sets the partial's return value; it does not discard template text the partial has already emitted, so a guard written that way still prints the empty shell. Hence the emptiness test wraps the markup instead of preceding it. */ -}} {{- /* `default dict` keeps the lookups below safe on a page with no `related` at all, so the whole computation stays flat. */ -}} {{- $rel := .Params.related | default dict -}} {{- $seeAlso := or $rel.seealso slice -}} {{- $backlinks := or $rel.backlinks slice -}} {{- $mentions := or $rel.mentions slice -}} {{- /* Merge both directions, deduped by URL. `union` can't hash slices of maps, so we walk the two lists in order and dedupe by URL manually. Deliberate references take priority; mentions precede backlinks among inferred connections. */ -}} {{- $seen := slice -}} {{- $merged := slice -}} {{- $deliberate := slice -}} {{- range $seeAlso -}} {{- if not (in $seen .url) -}} {{- $seen = $seen | append .url -}} {{- $deliberate = $deliberate | append . -}} {{- end -}} {{- end -}} {{- range $mentions -}} {{- if not (in $seen .url) -}} {{- $seen = $seen | append .url -}} {{- $merged = $merged | append . -}} {{- end -}} {{- end -}} {{- range $backlinks -}} {{- if not (in $seen .url) -}} {{- $seen = $seen | append .url -}} {{- $merged = $merged | append . -}} {{- end -}} {{- end -}} {{- /* Plural, human-readable label for each type. Falls back to `humanize` for any type not listed here. */ -}} {{- $labels := dict "affiliation" "Affiliations" "being" "Beings" "blog-post" "Posts" "character" "People" "continent" "Continents" "creature" "Creatures" "faith" "Faiths" "language" "Languages" "lore" "Lore" "organization" "Organizations" "page" "Pages" "pantheon" "Pantheons" "people" "Peoples" "polity" "Polities" "project-page" "Projects" "region" "Regions" "religion" "Religions" "settlement" "Settlements" "site" "Places" "world" "Worlds" -}} {{- /* Group refs by type, preserving alphabetical order within a group (the input lists arrive already sorted). */ -}} {{- $groups := dict -}} {{- range $merged -}} {{- $t := .type | default "page" -}} {{- $existing := index $groups $t | default slice -}} {{- $groups = merge $groups (dict $t ($existing | append .)) -}} {{- end -}} {{- /* Sort the group keys alphabetically by their rendered label so the list reads A → Z regardless of insertion order. */ -}} {{- $rows := slice -}} {{- range $type, $refs := $groups -}} {{- $label := index $labels $type | default (humanize $type) -}} {{- $rows = $rows | append (dict "label" $label "type" $type "refs" $refs) -}} {{- end -}} {{- if $rows -}} {{- $rows = sort $rows "label" "asc" -}} {{- end -}} {{- if or $rows $deliberate -}}
